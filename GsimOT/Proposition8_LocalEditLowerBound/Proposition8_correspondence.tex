\begin{center}{\Large\bfseries Proposition 8: no linear bound for the local edit distance}\\[4pt]

{\small Correspondence between the paper and the Lean~4/Mathlib formalisation}\end{center}

\noindent\textbf{Paper.} J.~Breitung, T.~Gervens, M.~Grohe. \emph{Graph Similarity Meets Optimal Transport} (file \texttt{Graph\_Similarity\_Meets\_Optimal\_Transport.pdf} in the repository). Statement, equation, page and line numbers refer to this PDF; \texttt{docs/NUMBERING.md} lists the Lean files and declarations of all numbered statements.\par\smallskip

The folder \texttt{GsimOT/Proposition8\_LocalEditLowerBound} formalises Proposition~8 together with the construction of Appendix~D.1 and Lemmas~27 and~28, Theorem~3~(2), and the remarks after Proposition~8 and Theorem~9 that concern $G_r,H_r$. Shared notions and lemmas (Observation~21, Lemma~22, Lemma~29) come from \texttt{GsimOT/Common}; they are listed at the end.\par\paragraph{Checking.} Run \texttt{bash build.sh} in the repository root (see \texttt{README.md}). The script builds the library and runs \texttt{scripts/CheckAxioms.lean}, which must report only the standard axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}: no \texttt{sorry} and no additional axioms.

\paragraph{Main statement in Lean.} \lean{GsimOT.Prop8.proposition8}: for every $r\in\mathbb N$ with $2\le r$ and $r$ even,
\[ d_1(G_r,H_r)=r,\qquad d_1^{\mathrm{OT}}(G_r,H_r)=1,\qquad r^3\le|V(G_r)|\le 19\,r^3 . \]
\lean{GsimOT.Prop8.no_linear_bound}: for every $c\in\mathbb R$ there is an even $r\ge2$ with
$\delta_1(G_r,H_r)>c\cdot\delta_1^{\mathrm{OT}}(G_r,H_r)$.
\lean{GsimOT.Prop8.no_power_bound}: for all $C\in\mathbb R$ and $\alpha>2/3$ there is an even $r\ge2$ with
$\delta_1(G_r,H_r)>C\cdot(\delta_1^{\mathrm{OT}}(G_r,H_r))^\alpha$.

\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item Proposition~8 states $\Theta(r^3)$ vertices. The formalisation proves $r^3\le|V(G_r)|\le19r^3$ and, more precisely, $|V(G_r)|=5r^3+10r^2-r+4$ (\lean{card_Vr}).
\item Indices are 0-based in Lean: \texttt{Marker.clock j h} is $c_{j+1,h+1}$, \texttt{Marker.tag i h} is $t_{i+1,h+1}$, \texttt{Marker.bit i} is $b_{i+1}$ and \texttt{Core.v i k} is $v_{i+1,k+1}$.
\item Adjacency is defined through the pattern \lean{pat} $r\,i\,t$ of branch $i$ at time $t\in[0,\ell]$, which is exactly the table of Appendix~D.1 (p.~26, l.~800) with $k$ replaced by $t$. Thus $\chi(v_{i,k})=\mathtt{pat}\,i\,k$, $\chi_{G_r}(u)=\mathtt{pat}\,i\,0=0$ and $\chi_{H_r}(u)=\mathtt{pat}\,i\,\ell=\mathbf 1_C$.
\item The proof of Lemma~27 obtains $d_1^{\mathrm{OT}}(G_r,H_r)\ge1$ from Lemma~29 since $G_r\not\cong H_r$ (p.~27, l.~837). The formalisation derives $G_r\not\cong H_r$ from Lemma~28: every bijection has cost $\ge r>0$, while an isomorphism would have cost $0$. (The proof idea, p.~8, l.~282--283, obtains the equality from Observation~21 instead: $u$ is isolated in $G_r$ but $H_r$ has no isolated vertex.)
\item In the proof of Lemma~27 (p.~27, l.~836--837) the paper computes the local costs at clock markers, tags and bits as exactly $1$, $2/r$ and $1$. The formalisation proves the upper bounds $1$, $2/r$ and $1$, which are all that is needed for $C_1(Q_r)\le1$.
\item Proof of Proposition~8: for $d_1(G_r,H_r)\le r$ the formalisation shows that \emph{every} local cost of $\sigma_1$ is at most $r$ (core pairs $\le1$, clocks $\le1$, tags $\le2$, bits $\le r$), which is the content of ``its local costs can be computed \dots\ as in the proof of Lemma~27''.
\item Proof of Lemma~28: the walk is formalised via the permutation $P$ of the core induced by $\pi$ and its orbit $u,P(u),\dots,P^{s}(u)=u$, with $s$ the least positive period. The pattern sequence is $x_\iota=\chi_{G_r}(P^\iota(u))$ for $\iota<s$ and $x_s=\chi_{H_r}(u)$, as in the paper. The indices $\alpha,\beta,\iota_j$ are defined exactly as in the paper (via \texttt{Nat.find} / \texttt{Nat.findGreatest}). For patterns on different branches the formalisation only uses Hamming distance $\ge r$ (the paper notes $\ge2r$), which suffices. At a core vertex $x$ it only uses that $L_\pi(x,\pi(x))$ is at least the Hamming distance of the patterns (the paper: equal to it, p.~27, l.~840--841).
\item The remark after Theorem~9 (p.~9, l.~318--320): since $\delta_1(G_r,H_r)=r/n$, $\delta_1^{\mathrm{OT}}(G_r,H_r)=1/n$ and $n=|V(G_r)|\le19r^3$, no bound $\delta_1\le C(\delta_1^{\mathrm{OT}})^\alpha$ with $\alpha>2/3$ holds (\lean{no_power_bound}). The exact asymptotics $\Theta(r^{-3})$, $\Theta(r^{-2})$ are not stated separately.
\end{enumerate}
