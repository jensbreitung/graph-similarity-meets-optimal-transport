\begin{center}{\Large\bfseries Theorem 6: upper bound for the edit distance}\\[4pt]

{\small Correspondence between the paper and the Lean~4/Mathlib formalisation}\end{center}

\noindent\textbf{Paper.} J.~Breitung, T.~Gervens, M.~Grohe. \emph{Graph Similarity Meets Optimal Transport} (file \texttt{Graph\_Similarity\_Meets\_Optimal\_Transport.pdf} in the repository). Statement, equation, page and line numbers refer to this PDF; \texttt{docs/NUMBERING.md} lists the Lean files and declarations of all numbered statements.\par\smallskip

The folder \texttt{GsimOT/Theorem6\_EditUpperBound} formalises Theorem~6 (p.~7) with its proof in Appendix~C.2 (Eq.~(17) and the steps (a)--(g)) and the $\delta_{\mathrm{ed}}$ part of Theorem~2. Shared notions (the edit distance, Eq.~(3)) come from \texttt{GsimOT/Common}; they are listed at the end.\par\paragraph{Checking.} Run \texttt{bash build.sh} in the repository root (see \texttt{README.md}). The script builds the library and runs \texttt{scripts/CheckAxioms.lean}, which must report only the standard axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}: no \texttt{sorry} and no additional axioms.

\paragraph{Main statement in Lean.} \lean{GsimOT.theorem6}: for weights $\lambda_i\ge0$ with $\sum_i\lambda_i=1$ and bijections
$\pi_i\colon V\to W$ over a finite index set, if $Q=\sum_i\lambda_iP_{\pi_i}$, then
\[ \sum_i\lambda_i\,C_{\mathrm{ed}}(\pi_i)\le2\,C_{\mathrm{ed}}(Q). \]
\lean{GsimOT.theorem6_dEdit}: $d_{\mathrm{ed}}(G,H)\le2\,d^{\mathrm{OT}}_{\mathrm{ed}}(G,H)$, and
\lean{GsimOT.theorem2_edit} (Theorem~2, first bound): $\delta_{\mathrm{ed}}(G,H)\le2\,\delta^{\mathrm{OT}}_{\mathrm{ed}}(G,H)$,
for all finite graphs $G,H$.

\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item The formalisation follows the proof of Theorem~6 in Appendix~C.2. The expectation $\mathbb E[f(\pi)]$ over the random bijection with $\Pr(\pi=\pi_i)=\lambda_i$ is the sum $\sum_i\lambda_if(\pi_i)$, and the chain of l.~767--777 is a \texttt{calc} block with one step per line, commented with the paper's line numbers and labels (a)--(g).
\item Matrices are functions $V\to W\to\mathbb R$, as in \texttt{Common/Basic.lean}: \lean{adjMul G R} is $A_GR$, \lean{mulAdj H R} is $RA_H$, \lean{conjMul H R} is $RA_HR^\top$ and \lean{frob X Y} is $\langle X,Y\rangle_F$, so $\|X\|_F^2$ is \lean{frob X X}.
\item A decomposition is given by weights $\lambda_i\ge0$ with $\sum_i\lambda_i=1$ over any finite index type; only these are assumed, so \lean{theorem6} applies to every decomposition as in Eq.~(3).
\item The first step of the chain (l.~767) holds since the entries of $A_G$ and $A_H$ are in $\{0,1\}$ (\lean{costEditBij_eq_frob_conjMul}).
\item Example~7 (p.~7) and the remark at the end of Appendix~C.2 (p.~25, arbitrary matrices via thresholds) are not formalised.
\end{enumerate}
