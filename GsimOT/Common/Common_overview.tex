\noindent Every result of the formalisation is stated in terms of the definitions in this folder. A wrong
definition would make the theorems meaningless, so this overview lists \emph{every} notion of the paper that
the Lean statements use, next to its Lean definition, and explains each representation choice. It then
lists the shared lemmas and which result folders use them.

\section{Conventions}
\begin{itemize}
\item \textbf{Vertex types.} A graph is a Mathlib \lean{SimpleGraph V} on an arbitrary finite type \lean{V}
  (\lean{[Fintype V] [DecidableEq V]}) with decidable adjacency. Simple, undirected and loopless are built into
  \lean{SimpleGraph}. The paper's ``graphs $G,H$ with vertex sets $V,W$'' are \lean{G : SimpleGraph V},
  \lean{H : SimpleGraph W}. Bijections $\pi\in\mathrm{Bij}(V,W)$ are equivalences \texttt{V $\simeq$ W}.
\item \textbf{Matrices} $Q\in\mathbb R^{V\times W}$ are functions \texttt{V $\to$ W $\to$ $\mathbb{R}$}; matrices indexed by
  $V\times W$ (such as $B$) are \texttt{Matrix (V $\times$ W) (V $\times$ W) $\mathbb{R}$}. All entries are real numbers; the adjacency
  matrix is real-valued with entries in $\{0,1\}$.
\item \textbf{Minima are infima.} Every $\min$ in the definitions of $d_\diamondsuit$, $d^{\mathrm{OT}}_\diamondsuit$,
  $\delta^{\mathrm{OT}}_{p\to q}$ is a Lean infimum (\texttt{iInf}). For the finite set $\mathrm{Bij}(V,W)$ this is the minimum.
  For optimal transport the paper argues that the minima are attained (Appendix~A, pp.~14--15, l.~503--510); this is
  formalised for the operator norms (\lean{exists_deltaOTPQ_eq}, \lean{exists_dOT_eq}) and the edit distance
  (\lean{exists_dOTEdit_eq}); for $d^{\mathrm{OT}}_1$ and $d^{\mathrm{OT}}_\square$ it is not needed. Infima in
  $\mathbb{R}$ of empty families are $0$: if $|V|\neq|W|$ there is no bijection and no doubly stochastic matrix, and
  $d_\diamondsuit$, $d^{\mathrm{OT}}_\diamondsuit$ evaluate to $0$ (the paper does not define them in this case).
  Results that need $|V|=|W|$ or $V,W\neq\emptyset$ assume it explicitly.
\item \textbf{Convention $0^0=0$} (p.~3, l.~84; Appendix~A, p.~14, l.~500) is built into \lean{powZ x a} $=0$ if $x=0$, else $x^a$.
\item \textbf{Operator norms} are suprema over the unit \emph{ball}; for nonempty index sets this equals the
  paper's maximum over the unit sphere (Appendix~A, p.~14, l.~479). Lemma~19 is stated with unit balls as well.
\item \textbf{Normalisation.} For graphs of the same order $n$, $\delta_\diamondsuit=d_\diamondsuit/s(n)$ and
  $\delta^{\mathrm{OT}}_\diamondsuit=d^{\mathrm{OT}}_\diamondsuit/s(n)$, with $s(n)=n^2$ for the edit and the cut distance
  (p.~1, l.~24--27) and $s(n)=n^{\kappa_{p\to q}}$, $\kappa_{p\to q}=1/q+1-1/p$, for $\|\cdot\|_{p\to q}$
  (p.~3, l.~88--90; Appendix~B, p.~15, l.~513--514); so $s(n)=n$ for the $p$-operator norms $\|\cdot\|_p=\|\cdot\|_{p\to p}$.
  Lean divides by $n=|V|$ or $n^2$ (\lean{deltaOne}, \lean{deltaOTOne}, \lean{deltaEdit}, \lean{deltaOTEdit},
  \lean{deltaCut}, \lean{deltaOTCut}) and multiplies by $n^{-\kappa}$ in \lean{deltaPQ}; \lean{deltaOTPQ_eq_dOT} proves
  $\delta^{\mathrm{OT}}_{p\to q}=n^{-\kappa}d^{\mathrm{OT}}_{p\to q}$ from $\mathrm{DS}(V,W)=n\,\mathrm{DS}^*(V,W)$ (Appendix~A, p.~14, l.~494--495).
\end{itemize}

\section{The shared lemmas: paper versus Lean}
\begin{itemize}
\item \textbf{Lemma 19} (p.~14, l.~488--492; $\|A\|_{p\to q}=\max_{\|y\|_{q^*}=1,\|x\|_p=1}|y^\top Ax|$): stated with unit balls; proved via
  H\"older (\lean{abs_dotProduct_le_lpNorm_mul}) and its equality case (\lean{exists_dotProduct_eq_lpNorm}), as in the paper.
\item \textbf{Eq.~(3)} (p.~5, l.~193--196): \lean{exists_birkhoff_decomposition_pos} is the exact statement ($\lambda_i>0$, $\sum\lambda_i=1$).
  \lean{exists_birkhoff_decomposition} indexes by all permutations with $\lambda\ge0$, which
  Theorems~6 and 14 use. All are derived from Mathlib's Birkhoff--von Neumann theorem.
\item \textbf{Eq.~(7)}: $\|B_{1\to1}(G,H,Q)\|_{1\to1}=\max_{\mathrm{supp}(Q)}L_Q$ for $Q\in\mathrm{DS}(V,W)$, the identity
  $C_1(Q)=\max_{(v,w)\in\mathrm{supp}(Q)}L_Q(v,w)$ of Section~2 (p.~5, l.~188--189). Eq.~(7) (p.~14, l.~500--502) is the same
  identity for couplings $Q^*\in\mathrm{DS}^*(V,W)$, inside the minimum defining $\delta^{\mathrm{OT}}_1$.
\item \textbf{Appendix~A, minima attained} (pp.~14--15, l.~503--510): the paper's argument (compactness; for operator norms, restricting
  $B_{p\to q}(G,H,Q_k)$ to the rows and columns indexed by $\mathrm{supp}(\lim Q_k)$ gives convergent matrices and does
  not increase the norm) is \lean{opNorm_restrict_le} and \lean{lowerSemicontinuous_opNorm_Bmat}.
\item \textbf{Observation 21} (p.~20, l.~642--643; $L_Q(v,w)\ge|\deg_G(v)-\deg_H(w)|$): triangle inequality and the row/column sums of $Q$, as in the paper.
\item \textbf{Lemma 22} (locking markers, pp.~20--21, l.~650--674): new vertex set $V\oplus[m]$, clique vertex $y_{i+1}$ = \lean{Sum.inr i}, marker
  index $j$ = \lean{j.val + 1}; $\Delta,\tau,\delta$ are natural numbers (the paper: integers $\tau\ge1$, $\delta\ge0$). Both items
  for $\diamondsuit=1$ follow the paper; for $\diamondsuit=\mathrm{ed}$, item~1 is derived
  from $\diamondsuit=1$ via $C_1(\pi)\le C_{\mathrm{ed}}(\pi)$ and item~2 is proved directly.
\item \textbf{Lemma 29} (p.~28, l.~872--886): stated for any $Q\in\mathrm{DS}(V,W)$ whose supported local costs are at most $L$ (the paper: an
  optimal $Q$ and $L=d_1^{\mathrm{OT}}$), with a bijection $\sigma$ inside $\mathrm{supp}(Q)$ (Eq.~(18)).
\end{itemize}
