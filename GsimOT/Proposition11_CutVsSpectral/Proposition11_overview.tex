\section{Statement}

\paragraph{Paper.} Proposition~12 (Section~3.4, p.~11, l.~385--386): \emph{for all graphs $G,H$ of the same order,
$\delta_\square(G,H)\le\delta_2(G,H)\le2\sqrt{\delta_\square(G,H)}$.} ``The second inequality uses a result of
Nikiforov [Nik09]'' (l.~387). The proof is at the beginning of App.~F (pp.~44--45, l.~1382--1397). It shows, for all
$A\in\mathbb R^{V\times W}$ with $m=|V|$, $n=|W|$ (p.~44, l.~1384),
\[ \tfrac1{\sqrt{mn}}\|A\|_\square\le\|A\|_2\le2\sqrt{\|A\|_{(\infty)}\|A\|_\square},
   \qquad \|A\|_{(\infty)}:=\max_{v\in V,w\in W}|A_{vw}| \ \text{(l.~1385)}, \]
the first by Cauchy--Schwarz and the second by citing Nikiforov [Nik09, Theorem~3], and applies it to
$A^\pi_G-A_H$, whose entries lie in $[-1,1]$, with the normalisations $\delta_\square=n^{-2}d_\square$,
$\delta_2=n^{-1}d_2$ (l.~1385--1387).

\paragraph{Lean.} \lean{GsimOT.proposition11}: for finite types $V,W$ with \hyp{$|V|=|W|$} and simple graphs
$G,H$,
\[ \delta_\square(G,H)\le\delta_2(G,H)\quad\text{and}\quad\delta_2(G,H)\le2\sqrt{\delta_\square(G,H)}, \]
with $\delta_\square=$ \lean{deltaCut G H} and $\delta_2=$ \lean{deltaPQ 2 2 G H}. This is exactly the statement.
The matrix inequalities are \lean{cutNorm_le_sqrt_mul_opNorm_two} ($\|A\|_\square\le\sqrt{mn}\,\|A\|_2$) and
\lean{opNorm_two_le_two_sqrt_cutNorm} (Nikiforov's inequality, \emph{proved} here).

\section{The proof step by step}

\paragraph{First inequality (pp.~44--45, l.~1388--1396).} $\|A\|_\square=\max_{S,T}|\langle\mathbf 1_S,A\mathbf 1_T\rangle|
\le\max\|\mathbf 1_S\|_2\|A\mathbf 1_T\|_2\le\sqrt{mn}\,\|A\|_2$, with $\|\mathbf 1_U\|_2\le\sqrt{|U|}$.
\lean{cutNorm_le_sqrt_mul_opNorm_two} follows this: the indicator vector \lean{ind}, $\|A\mathbf 1_T\|_2\le
\|A\|_2\sqrt n$ (\lean{lpNorm_mulVec_le}), and Cauchy--Schwarz \texttt{sq\_sum\_le\_card\_mul\_sum\_sq} for the sum
over $S$.

\paragraph{Second inequality (p.~45, l.~1397, cited).} The paper cites [Nik09, Theorem~3]. The Lean proves the version
with an arbitrary entrywise bound $M$, \lean{opNorm_two_le_two_sqrt_of_abs_le}: by the Schur test,
$\|Ax\|_2^2=x^\top(A^\top A)x\le\max_j\sum_k|(A^\top A)_{jk}|\cdot\|x\|_2^2$, and
$\sum_k|(A^\top A)_{jk}|=\sum_iA_{ij}(As)_i\le M\|A\|_{\infty\to1}\le4M\|A\|_\square$ for a sign vector $s$
(\lean{infOneNorm_le_four_cutNorm}: split sign vectors into positive and negative parts, giving four cut values).
\lean{opNorm_two_le_two_sqrt_cutNorm} is the case $M=\|A\|_{(\infty)}$.

\paragraph{Application to graphs (p.~44, l.~1385--1387).} In \lean{proposition11}: for the first inequality take a
bijection $\pi$ minimising $\|A^\pi_G-A_H\|_2$; $d_\square\le\|A^\pi_G-A_H\|_\square\le n\,d_2$, i.e.
$\delta_\square=d_\square/n^2\le d_2/n=\delta_2$. For the second take $\pi$ minimising the cut norm; $M=1$ by
\lean{abs_adj_sub_adj_le_one}, so $d_2\le2\sqrt{d_\square}$, i.e. $\delta_2\le2\sqrt{\delta_\square}$. Minima over
the finite set of bijections exist (\texttt{exists\_eq\_ciInf\_of\_finite}); $n=0$ is trivial.

\section{Notes on the formalisation}
\begin{itemize}
\item Nikiforov's inequality [Nik09, Thm.~3] is proved instead of cited, in the more general form with an arbitrary
  entrywise bound $M$ (\lean{opNorm_two_le_two_sqrt_of_abs_le}); the application to graphs uses $M=1$.
\item The first matrix inequality is stated as $\|A\|_\square\le\sqrt{mn}\,\|A\|_2$
  (\lean{cutNorm_le_sqrt_mul_opNorm_two}), the paper's $\frac1{\sqrt{mn}}\|A\|_\square\le\|A\|_2$ multiplied by $\sqrt{mn}$.
\end{itemize}
