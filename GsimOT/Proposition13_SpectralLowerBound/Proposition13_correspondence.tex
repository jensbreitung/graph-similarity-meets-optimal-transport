\begin{center}{\Large\bfseries Proposition 13: no linear bound for the spectral distance}\\[4pt]

{\small Correspondence between the paper and the Lean~4/Mathlib formalisation}\end{center}

\noindent\textbf{Paper.} J.~Breitung, T.~Gervens, M.~Grohe. \emph{Graph Similarity Meets Optimal Transport} (file \texttt{Graph\_Similarity\_Meets\_Optimal\_Transport.pdf} in the repository). Statement, equation, page and line numbers refer to this PDF; \texttt{docs/NUMBERING.md} lists the Lean files and declarations of all numbered statements.\par\smallskip

The folder \texttt{GsimOT/Proposition13\_SpectralLowerBound} formalises Proposition~13 with the construction of Appendix~F.1, Lemma~49, Theorem~3~(4) and the remark after Proposition~13.\par\paragraph{Checking.} Run \texttt{bash build.sh} in the repository root (see \texttt{README.md}). The script builds the library and runs \texttt{scripts/CheckAxioms.lean}, which must report only the standard axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}: no \texttt{sorry} and no additional axioms.

\paragraph{Main statement in Lean.} \lean{GsimOT.Prop13.proposition13}: for every $r\ge1$, with $n=|V(G_r)|=4r^2+3r+1$,
\[ d_2(G_r,H_r)=\sqrt r\qquad\text{and}\qquad \tfrac{1}{2\sqrt2}\le d^{\mathrm{OT}}_2(G_r,H_r)\le1; \]
\lean{GsimOT.Prop13.proposition13_explicit} adds $\frac1{2\sqrt2}\le\frac r{\sqrt n}\le d^{\mathrm{OT}}_2(G_r,H_r)$ (Lemma~49).
\lean{GsimOT.Prop13.no_power_bound}: for every $\alpha>\frac34$ and $C$, some $G_r,H_r$ violate $\delta_2\le C(\delta^{\mathrm{OT}}_2)^\alpha$.

\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item \textbf{Upper bound of Lemma~49.} The paper computes $C_2(Q)^2=\|K^\top K\|_2=1$ for the block form $\begin{pmatrix}0&K\\K^\top&0\end{pmatrix}$. The formalisation instead shows $\sum_{(x,y)}Q(x,y)(d_{xy}\cdot\zeta)^2\le\|\zeta\|^2$. It uses Cauchy--Schwarz on each tag block and the orthogonality of the clock markers, and concludes $|y^\top Bx|\le1$ through Lemma~19. This is the same estimate, organised as a bilinear bound.
\item The lower bound of Lemma~49 is proved for arbitrary graphs (\lean{le_dOT_of_deg}: a vertex whose degree differs from every degree in $H$ by $\ge c$ forces $d^{\mathrm{OT}}_2\ge c/\sqrt n$) and then specialised to $c=r$.
\item The remark after Proposition~13 is formalised as \lean{no_power_bound}. It uses $\delta_2(G_r,H_r)=\sqrt r/n$, $\delta^{\mathrm{OT}}_2(G_r,H_r)\le1/n$ and $n\le8r^2$; the lower estimate $\delta^{\mathrm{OT}}_2=\Omega(n^{-1})$ is not needed for this conclusion.
\item Indices: $v_{i,k}$ is \texttt{some (i, k)} and $u$ is \texttt{none} in \lean{Core}; clocks and tags are the two summands of \lean{Marker}. $G_r$ and $H_r$ are one construction \lean{graph} with a Boolean flag for the edges at $u$.
\item \lean{proposition13} is Proposition~13. \lean{proposition13_explicit} states Proposition~13 together with the bound $\frac r{\sqrt n}\le d^{\mathrm{OT}}_2(G_r,H_r)$ of Lemma~49, which the proof of Proposition~13 combines with $\frac r{\sqrt n}\ge\frac1{2\sqrt2}$; \lean{proposition13} is derived from it.
\end{enumerate}
