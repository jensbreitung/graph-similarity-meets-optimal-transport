\begin{center}{\Large\bfseries Theorem 1: blowup distances equal optimal transport distances}\\[4pt]

{\small Correspondence between the paper and the Lean~4/Mathlib formalisation}\end{center}

\noindent\textbf{Paper.} J.~Breitung, T.~Gervens, M.~Grohe. \emph{Graph Similarity Meets Optimal Transport} (file \texttt{Graph\_Similarity\_Meets\_Optimal\_Transport.pdf} in the repository). Statement, equation, page and line numbers refer to this PDF; \texttt{docs/NUMBERING.md} lists the Lean files and declarations of all numbered statements.\par\smallskip

The folder \texttt{GsimOT/Theorem1\_BlowupEqualsOT} formalises Theorem~1 and its proof in App.~B, including Claim~20. The operator norms, Lemma~19 and the attainment of the minimum in Eq.~(2) live in \texttt{GsimOT/Common} (\texttt{OperatorNorm.lean}, \texttt{Attainment.lean}). A proof overview is \texttt{Theorem1\_overview.pdf}.\par\paragraph{Checking.} Run \texttt{bash build.sh} in the repository root (see \texttt{README.md}). The script builds the library and runs \texttt{scripts/CheckAxioms.lean}, which must report only the standard axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}: no \texttt{sorry} and no additional axioms.

\paragraph{Main statement in Lean.} \lean{GsimOT.theorem1}: for all $p,q\in[1,\infty]$ and all finite graphs $G$ on $V\neq\emptyset$, $H$ on
$W\neq\emptyset$, \[ \delta^{\bullet}_{p\to q}(G,H)=\delta^{\mathrm{OT}}_{p\to q}(G,H). \]
\lean{GsimOT.theorem1_tendsto}: moreover $\delta_{p\to q}(G^{\bullet n\ell},H^{\bullet m\ell})\to
\delta^{\mathrm{OT}}_{p\to q}(G,H)$ as $\ell\to\infty$, where $m=|V|$, $n=|W|$ (so the limit exists).
\lean{GsimOT.theorem1_of_tendsto}: the paper's argument, assuming the existence of the limit.

\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item \textbf{Two versions of the final step.} The paper takes the existence of the limit $\delta^\bullet=\lim_\ell\delta_{p\to q}(G^{\bullet n\ell},H^{\bullet m\ell})$ from [Gro25] (p.~2) and proves the upper bound only along the subsequence $\ell=ks$ (Eq.~(15)). \lean{theorem1_of_tendsto} is this argument verbatim, with the existence of the limit as a hypothesis. The main theorem \lean{theorem1} needs no such hypothesis: its approximation step (\lean{exists_integral_approx}) works for \emph{every} sufficiently large $\ell$, so the whole sequence converges to $\delta^{\mathrm{OT}}_{p\to q}$ (\lean{theorem1_tendsto}). In Lean, $\delta^\bullet$ is \texttt{limUnder} of the sequence (\lean{deltaBlowup}).
\item \textbf{Claim 20} (p.~17) is formalised as \lean{claim20} and follows the paper: a minimiser $Q^*_0$ (\lean{exists_deltaOTPQ_eq}, from the attainment argument of App.~A), rational couplings $Q^*_k\to Q^*_0$ with the same support, and convergence of the norms on a fixed support. The rational sequence is obtained differently: the paper uses that the couplings vanishing outside $S$ form a rational polytope; we construct the sequence explicitly. $Q^*_0$ is lifted to the doubly stochastic $mn\times mn$ matrix $R((v,i),(w,j))=Q^*_0(v,w)$, decomposed by Birkhoff--von Neumann (Mathlib), the positive coefficients are rounded to positive multiples of $1/\ell$, and the result is averaged back. This preserves the support, which is essential: with $0^0=0$, $Q\mapsto B_{p\to q}(Q)$ is discontinuous at the boundary of the support for $p=1$ or $q=\infty$.
\item Eqs.~(11) and (15) follow the paper's argument verbatim: the coupling $Q^*=|J|/N$, Eq.~(10), maximising test vectors from Lemma~19, the spread (resp.\ averaged) vectors and the chain of equalities. The norm computations for these vectors are factored out into the general block lemmas \lean{lpNorm_spread_le} and \lean{lpNorm_average_le}, for a block map $b:\beta\to\alpha$ with blocks $\{w\}\times J(v,w)$. \lean{eq13} is Eq.~(11) and \lean{eq16} is Eq.~(15).
\item \textbf{Normalisation.} $\delta_{p\to q}=N^{-\kappa_{p\to q}}d_{p\to q}$ for graphs of order $N$, i.e.\ the metric is scaled by $1/s(N)$ with $s(N)=N^{1/q+1-1/p}$ (p.~3, l.~88--90; App.~B, p.~15, l.~513--514, and p.~20, l.~637--638).
\item Operator norms are defined as the supremum of $\|Ax\|_q$ over the unit \emph{ball} $\|x\|_p\le1$, and Lemma~19 is stated with $\|x\|_p\le1$, $\|y\|_{q^*}\le1$. For nonempty index sets this equals the paper's maximum over unit spheres by homogeneity; the paper's proof notes this itself (p.~20, l.~636--637).
\item Theorem~1 is formalised for general $p,q\in[1,\infty]$ (\texttt{ENNReal}). It is independent of the local edit distance development: for $p=q=1$ it concerns $\|B_{1\to1}\|_1$, which the paper relates to $C_1$ in Eq.~(7); this link is proved in Common (\lean{dOT_one_one_eq_dOTOne}) but not needed here.
\end{enumerate}
