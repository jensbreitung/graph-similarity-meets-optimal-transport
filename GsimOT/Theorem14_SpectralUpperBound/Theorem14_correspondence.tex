\begin{center}{\Large\bfseries Theorem 14: upper bound for the spectral distance}\\[4pt]

{\small Correspondence between the paper and the Lean~4/Mathlib formalisation}\end{center}

\noindent\textbf{Paper.} J.~Breitung, T.~Gervens, M.~Grohe. \emph{Graph Similarity Meets Optimal Transport} (file \texttt{Graph\_Similarity\_Meets\_Optimal\_Transport.pdf} in the repository). Statement, equation, page and line numbers refer to this PDF; \texttt{docs/NUMBERING.md} lists the Lean files and declarations of all numbered statements.\par\smallskip

The folder \texttt{GsimOT/Theorem14\_SpectralUpperBound} formalises Theorem~14, its proof in Appendix~F.2 with Lemmas~50--52, and the $\delta_2$ part of Theorem~2. No cited results are assumed.\par\paragraph{Checking.} Run \texttt{bash build.sh} in the repository root (see \texttt{README.md}). The script builds the library and runs \texttt{scripts/CheckAxioms.lean}, which must report only the standard axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}: no \texttt{sorry} and no additional axioms.

\paragraph{Main statement in Lean.} \lean{GsimOT.theorem14}: for all graphs $G,H$ of the same order, with $\delta=\delta^{\mathrm{OT}}_2(G,H)$,
\[ \delta_2(G,H)\le\delta+2\sqrt{2\delta}. \]
\lean{GsimOT.theorem14_h}: $\delta_2(G,H)\le\delta+4\sqrt{h\delta}$ with $h=\min\{\sqrt{p_G(1-p_G)},\sqrt{p_H(1-p_H)}\}$ (end of the proof);
\lean{GsimOT.theorem14_min}: $\delta_2(G,H)\le\min\{1,\delta+4\sqrt{h\delta}\}\le\min\{1,\delta+2\sqrt{2\delta}\}$ (with $\delta_2\le1$);
\lean{GsimOT.theorem14_sqrt}: $\delta_2(G,H)\le(1+2\sqrt2)\sqrt{\delta}$.

\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item \textbf{Statement.} \lean{theorem14} is Theorem~14, $\delta_2\le\delta^{\mathrm{OT}}_2+2\sqrt{2\delta^{\mathrm{OT}}_2}$. As in the paper, it follows from the bound $\delta_2\le\gamma_Q+4\sqrt{\gamma_Qh}$ at the end of the proof (\lean{theorem14_h}) and $h\le\frac12$. \lean{theorem14_min} combines \lean{theorem14_h} with $\delta_2\le1$ (\lean{deltaPQ_two_two_le_one}, via $\|\cdot\|_2\le\|\cdot\|_F\le n$) into $\delta_2\le\min\{1,\delta^{\mathrm{OT}}_2+4\sqrt{h\,\delta^{\mathrm{OT}}_2}\}\le\min\{1,\delta^{\mathrm{OT}}_2+2\sqrt{2\delta^{\mathrm{OT}}_2}\}$, which implies Theorem~14.
\item \textbf{Proof structure.} The formalisation follows Appendix~F.2 step by step: \lean{lemma50} uses Lemma~19 and the bilinear bound \lean{eq40_le} behind Eq.~(40) (two test vectors); \lean{eq42} and \lean{lemma51} follow the proof of Lemma~51 (via $F_{ij}$ and the expansion $\sum Q(F-m)^2=\sum QF^2-\sum m^2$); Eq.~(43) is \lean{eq43} (expectation of the squared spectral norm, bounded by the Frobenius norm computed in the eigenbasis, \lean{frobenius_eq_sum_eigvec}); the chain (a) uses Eq.~(28) for $A'_t$ (\lean{eq28}) and Eq.~(41) (\lean{eq41}); and the proof starts from an optimal $Q$ (\lean{exists_dOT_eq}).
\item \textbf{No symmetry reduction.} Instead of assuming $h=\sqrt{p_G(1-p_G)}$ by symmetry (l.~1529), both cases are proved: centre by $p_G$ and use $\|A'\|_F$ (\lean{deltaPQ_le_threshold_G}), or centre by $p_H$ and use $\|B'\|_F$, applying the argument to $(B',A',Q^\top)$ (\lean{deltaPQ_le_threshold_H}). The minimum is then taken.
\item \textbf{Normalisations.} The Lean works with $\|B_{2\to2}(G,H,Q)\|_2=n\gamma_Q$ (\lean{discKernel}) and with the threshold $s=nt$; all statements are the paper's after division by $n$. Expectations over the random permutation are weighted sums over a Birkhoff decomposition (\lean{exists_birkhoff_decomposition}).
\item \textbf{$\delta^{\mathrm{OT}}_2\le1$} (used after Theorem~14, l.~399) is proved as \lean{deltaOTPQ_two_two_le_one}: $\|B_{2\to2}(G,H,Q^*)\|_2\le\|B_{2\to2}(G,H,Q^*)\|_F\le\sum Q^*\le1$.
\end{enumerate}
