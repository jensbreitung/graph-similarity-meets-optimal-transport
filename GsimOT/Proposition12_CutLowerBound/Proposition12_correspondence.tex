\begin{center}{\Large\bfseries Proposition 10: a factor 2 gap for the cut distance}\\[4pt]

{\small Correspondence between the paper and the Lean~4/Mathlib formalisation}\end{center}

\noindent\textbf{Paper.} J.~Breitung, T.~Gervens, M.~Grohe. \emph{Graph Similarity Meets Optimal Transport} (file \texttt{Graph\_Similarity\_Meets\_Optimal\_Transport.pdf} in the repository). Statement, equation, page and line numbers refer to this PDF; \texttt{docs/NUMBERING.md} lists the Lean files and declarations of all numbered statements.\par\smallskip

The folder \texttt{GsimOT/Proposition12\_CutLowerBound} formalises Proposition~10 (p.~9) with the construction of App.~E.1, Lemmas~33 and~34, the remark after Proposition~10 ($d^{\mathrm{OT}}_\square(G,H)=4$; its Lean certificate, linked in footnote~2 of the paper, is \texttt{OptimalAlignment.lean}) and Theorem~3~(3).\par\paragraph{Checking.} Run \texttt{bash build.sh} in the repository root (see \texttt{README.md}). The script builds the library and runs \texttt{scripts/CheckAxioms.lean}, which must report only the standard axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}: no \texttt{sorry} and no additional axioms.

\paragraph{Main statement in Lean.} \lean{GsimOT.Prop12.proposition12}: the graphs $G,H$ of App.~E.1 have $145$ vertices,
\[ d_\square(G,H)=8\qquad\text{and}\qquad d^{\mathrm{OT}}_\square(G,H)\le4, \]
exactly as Proposition~10 states. \lean{GsimOT.Prop12.dOTCut_eq_four}: $d^{\mathrm{OT}}_\square(G,H)=4$, the
Lean certificate of the remark after Proposition~10. \lean{GsimOT.Prop12.theorem4_item3}:
$\delta_\square(G,H)=2\,\delta^{\mathrm{OT}}_\square(G,H)$ (Theorem~3~(3)).

\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item \textbf{$d^{\mathrm{OT}}_\square(G,H)=4$.} Proposition~10 states $d^{\mathrm{OT}}_\square(G,H)\le4$. The remark after it (p.~10, l.~330) states equality, verified by ``one of our Lean certificates''; its footnote~2 links the file \texttt{OptimalAlignment.lean}, whose main result is \lean{dOTCut_eq_four}. Theorem~3~(3) (p.~4, l.~126) uses this equality. The paper refers to this file for the proof, so the rows ``step 1--5'' above follow the outline in its module docstring. For every $Q\in\mathrm{DS}(V,W)$ with $T$ the mass of $Q$ on the $20$ core atoms, $d(x)=\deg_G(v)-\deg_H(w)$ and $L=\sum_xq_x|d(x)|$: $4C_\square(Q)\ge L+B(s,t)\ge9(10-T)+\frac{16}5T-16\ge16$ for suitable test functions $s,t$.
\item The finite checks are done by \texttt{decide} (no \texttt{native\_decide}, so no extra axioms). $|x^\top Zy|\le8$ is reduced to $\sum_j|(Zx)_j|\le8$ over the $3^5$ vectors $x$. The lower bound $\ge8$ is checked for each of the $32$ sign patterns $r$ with an explicit table of maximisers \lean{wit}. This replaces the paper's case analysis on $|K|$ up to symmetry.
\item Both graphs live on the same vertex type \lean{Vtx} $=(\mathrm{Fin}\,5\times\mathrm{Bool})\oplus\Sigma_i\,\mathrm{Fin}(9(i+1))$; the natural bijection $\mathrm{id}$ of the paper is \texttt{Equiv.refl}.
\end{enumerate}
