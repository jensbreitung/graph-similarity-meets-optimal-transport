\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item \textbf{Nikiforov's inequality is proved, not cited.} The paper cites [Nik09, Thm.~3] (p.~45, l.~1397) for $\|A\|_2\le2\sqrt{\|A\|_{(\infty)}\|A\|_\square}$. The formalisation proves it directly. By the Schur test, $\|Ax\|_2^2=x^\top A^\top Ax\le\max_j\sum_k|(A^\top A)_{jk}|\,\|x\|_2^2$. Moreover $\sum_k|(A^\top A)_{jk}|=\sum_iA_{ij}(As)_i\le\|A\|_{(\infty)}\|A\|_{\infty\to1}\le4\|A\|_{(\infty)}\|A\|_\square$ for a suitable sign vector $s$.
\item $\delta_2=\delta_{2\to2}=n^{-\kappa_{2\to2}}d_2$ with $\kappa_{2\to2}=1$, using the shared definitions in \texttt{Common/AlignedDistances.lean} and the operator norms of \texttt{Common/OperatorNorm.lean}.
\item The first inequality is formalised as $\|A\|_\square\le\sqrt{mn}\,\|A\|_2$ (\lean{cutNorm_le_sqrt_mul_opNorm_two}), the paper's $\frac1{\sqrt{mn}}\|A\|_\square\le\|A\|_2$ (p.~44, l.~1384) multiplied by $\sqrt{mn}$.
\end{enumerate}
