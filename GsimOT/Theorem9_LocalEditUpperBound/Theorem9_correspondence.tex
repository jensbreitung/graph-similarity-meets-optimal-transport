\begin{center}{\Large\bfseries Theorem 9: upper bound for the local edit distance}\\[4pt]

{\small Correspondence between the paper and the Lean~4/Mathlib formalisation}\end{center}

\noindent\textbf{Paper.} J.~Breitung, T.~Gervens, M.~Grohe. \emph{Graph Similarity Meets Optimal Transport} (file \texttt{Graph\_Similarity\_Meets\_Optimal\_Transport.pdf} in the repository). Statement, equation, page and line numbers refer to this PDF; \texttt{docs/NUMBERING.md} lists the Lean files and declarations of all numbered statements.\par\smallskip

The folder \texttt{GsimOT/Theorem9\_LocalEditUpperBound} formalises Theorem~9 together with Appendix~D.2 (Lemmas~30, 31, Proposition~32 and the path-cutting construction); Theorem~9 is the $\delta_1$ part of Theorem~2. Shared notions and lemmas (Eq.~(3), Observation~21, Lemma~29) come from \texttt{GsimOT/Common}; they are listed at the end.\par\paragraph{Checking.} Run \texttt{bash build.sh} in the repository root (see \texttt{README.md}). The script builds the library and runs \texttt{scripts/CheckAxioms.lean}, which must report only the standard axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}: no \texttt{sorry} and no additional axioms.

\paragraph{Main statement in Lean.} \lean{GsimOT.theorem9}: assuming Lemma~30 (\lean{CVZRounding}), for all finite graphs $G$ on $V$
and $H$ on $W$ with $|V|=|W|$,
\[ \delta_1(G,H)\;\le\;15\sqrt{\delta_1^{\mathrm{OT}}(G,H)\cdot\ln\bigl(4/\delta_1^{\mathrm{OT}}(G,H)\bigr)} . \]
\lean{GsimOT.theorem9_power}: under the same hypotheses, for every $\alpha<1/2$ there is $C_\alpha$ with
$\delta_1(G,H)\le C_\alpha\,(\delta_1^{\mathrm{OT}}(G,H))^\alpha$.

\section*{Notes on the formalisation}

\begin{enumerate}[leftmargin=*]
\item \textbf{Lemma~30 is a hypothesis.} Lemma~30 is the dependent rounding theorem of Chekuri, Vondr\'ak and Zenklusen [CVZ11, Thm.~1.1], which the paper cites without proof. It is stated as the proposition \lean{CVZRounding} and appears as the explicit hypothesis \texttt{hCVZ} of \lean{theorem9} (no Lean \texttt{axiom} is introduced). It is specialised to doubly stochastic $Q$ (a point of the perfect matching polytope), which is the only case used, and a ``random matching'' is a finitely supported probability distribution on subsets of $V\times W$. Apart from this specialisation, the statement is the paper's (p.~29, l.~895--898).
\item Final estimate of the proof of Theorem~9 (p.~30, l.~945--947): the paper's bound $\frac{b}{\sqrt{fh}}\le4\sqrt{f/h}+\frac{2\sqrt2}{\sqrt h}+4\sqrt5\le\frac{4+2\sqrt2}{\sqrt{\ln4}}+4\sqrt5<15$ (using $f\le1$ and $h\ge\ln4$; the constant is $\approx14.74$) is formalised as \lean{final_estimate}.
\item In the proof of Theorem~9, the quantity $\sqrt{n/(2L)}$ is called $s$ in Lean, so $2\sqrt{2nL}=4Ls$.
\item Optimal vs.\ arbitrary $Q$: Lemmas~29, 31 and Proposition~32 are stated for any $Q\in\mathrm{DS}(V,W)$ whose supported local costs are at most $L$. The paper takes $Q$ optimal and $L=d_1^{\mathrm{OT}}$. Theorem~9 is first proved with $f=C_1(Q)/n$ for every $Q$ (\lean{theorem9_of_isDS}), and the statement for $\delta_1^{\mathrm{OT}}$ (an infimum in Lean) follows by continuity of $f\mapsto15\sqrt{f\ln(4/f)}$ at $f>0$. The case $d_1^{\mathrm{OT}}<1$ is the isomorphic case (Lemma~29). The paper distinguishes $L=0$ and $L>0$ and then uses $L\ge1$; the Lean proof splits directly into $L<1$ and $L\ge1$.
\item Convention $f=0$: in Lean $\ln0=0$ and $4/0=0$, so $15\sqrt{f\ln(4/f)}=0$ for $f=0$, matching the paper's convention.
\item Proof of Proposition~32: the path decomposition is formalised in two steps, exactly as described: (i)~the outgoing edge of every vertex at position $p\equiv\ell-1\pmod\ell$ on a path is removed (this cuts every path into consecutive blocks of $\ell$ vertices plus a final shorter block), and (ii)~every block is closed to a cycle. The bound ``number of closing edges $\le m+n/\ell$'' is proved by counting the $\le n/\ell$ cut positions (each owns the $\ell$ vertices of its block) plus the $m$ path ends. The paper counts $\sum_j\lceil p_j/\ell\rceil\le m+n/\ell$ instead; both give the same bound.
\item In Proposition~32 the bound $\rho_H(\sigma(v_k),\sigma(v_1))\le2(k-1)L$ is used in the weaker form $\le2\ell L$, as in the paper's statement of the claim.
\item Lemma~31 handles the degenerate case $n=0$ separately (the paper implicitly has $n\ge1$ since $L>0$).
\item The remark after Theorem~9 (p.~9, l.~320) that Theorem~9 yields $\delta_1\le C_\alpha(\delta_1^{\mathrm{OT}})^\alpha$ for every $\alpha<1/2$ is \lean{theorem9_power} (with $C_\alpha=15\sqrt{4^{\varepsilon}/\varepsilon}$, $\varepsilon=1-2\alpha$), derived from Theorem~9 via $\ln x\le x^{\varepsilon}/\varepsilon$ (\lean{sqrt_mul_log_le_rpow}).
\end{enumerate}
